  [ -z "${localsha:-}" ] && continue
  case "${remoteref:-}" in
    refs/heads/wip/*)
      case "$localsha" in
        *[!0]*) wip_refs=$((wip_refs+1)); wip_shas="$wip_shas $localsha" ;;
        *) wip_dels=$((wip_dels+1)) ;;
      esac ;;
    refs/owner-signoff/*)
      # Only a record takes the notes tier: a parentless commit, under its subject's 40-hex name, whose
      # tree is exactly the sign-off's seven plain files (record.txt plus each reviewer's request and
      # output), no directory among them. Anything else pushed here, including the old four-file shape
      # or a source commit, takes the full tier, as does a deletion.
      if [[ "${remoteref#refs/owner-signoff/}" =~ ^[0-9a-f]{40}$ ]] && [ "$(git rev-list --parents -n 1 "$localsha" 2>/dev/null)" = "$localsha" ] \
         && [ "$(git ls-tree --format='%(objectmode) %(objecttype) %(path)' "$localsha" 2>/dev/null | LC_ALL=C sort | tr '\n' ' ')" \
              = "100644 blob agy-request.txt 100644 blob agy.txt 100644 blob codex-request.txt 100644 blob codex.txt 100644 blob kimi-request.txt 100644 blob kimi.txt 100644 blob record.txt " ]; then
        note_refs=$((note_refs+1))
      else
        other_refs=$((other_refs+1))
        case "$localsha" in *[!0]*) full_shas="$full_shas $localsha" ;; esac
      fi ;;
    refs/notes/concrete-review|refs/notes/concrete-review-findings)
      case "$localsha" in
        *[!0]*) note_refs=$((note_refs+1)) ;;
        *) other_refs=$((other_refs+1)) ;;   # deleting a review note takes the full tier
      esac ;;
    *) other_refs=$((other_refs+1))
       case "$localsha" in *[!0]*) full_shas="$full_shas $localsha" ;; esac ;;
  esac
done <<< "$HOOK_STDIN"

if [ "$other_refs" -eq 0 ] && [ $((wip_refs + wip_dels)) -gt 0 ]; then
  bad=""
  for sha in $wip_shas; do
    rc=0; git grep -qE '^(<{7}|>{7}|\|{7})( |$)|^={7}$' "$sha" -- . ':(exclude)legacy/' >/dev/null 2>&1 || rc=$?
    case "$rc" in
      0) bad="$bad $sha" ;;
      1) ;;
      *) echo "pre-push (wip tier): could not scan $sha (git grep exit $rc) — refusing"; exit 1 ;;
    esac
  done
  [ -z "$bad" ] || { echo "pre-push (wip tier): conflict markers in$bad — refusing"; exit 1; }
  echo "pre-push: wip/* backup push ($wip_refs update(s), $wip_dels deletion(s)) — conflict-marker check only; the full set runs on ci/* and main"
  exit 0
fi
if [ "$other_refs" -eq 0 ] && [ "$note_refs" -gt 0 ]; then
  echo "pre-push: review-notes push (review notes, owner sign-offs) — no source tree, no gates"
  exit 0
fi

# The checks build sandbox repositories, and a linked worktree hands its hook
# GIT_DIR=<repo>/.git/worktrees/<name>: on 2026-11-06 that turned a review-tool check into writes on a
# real clone and on GitHub (ISS-0015). Every repository-making script now clears git's variables, and
# as a second wall a push that runs the checks must come from a full clone. The wip and notes tiers
# above run nothing, so a worktree may still back up its branch.
if [ "$(cd "$(git rev-parse --git-dir)" && pwd -P)" != "$(cd "$(git rev-parse --git-common-dir)" && pwd -P)" ]; then
  echo "pre-push: refusing: this push runs the root checks, and $ROOT_DIR is a linked worktree; push it from a full clone (git clone), never a worktree." >&2
  exit 1
fi

# The checks run in the gate shell (scripts/checks/lib/gate_env.sh refuses anywhere else). From
# outside it, the hook re-executes itself inside `nix develop` with the same arguments and stdin.
if [ "${CONCRETE_GATES_SHELL:-}" != 1 ] || ! command -v lake >/dev/null 2>&1; then
  if [ "${CONCRETE_PREPUSH_REENTERED:-0}" = 1 ]; then
    echo "pre-push: nix develop did not provide the gate shell with Lean — refusing to push." >&2; exit 1
  fi
  command -v nix >/dev/null 2>&1 || {
    echo "pre-push: the checks run inside nix develop, and nix is not on PATH — refusing to push." >&2; exit 1; }
  echo "pre-push: entering nix develop for the checks"
  CONCRETE_PREPUSH_REENTERED=1 exec nix develop "$ROOT_DIR" --command bash "$HOOK_SELF" "$@" <<< "$HOOK_STDIN"
fi

# The checks judge the work tree, so it must be the commit being pushed, unmodified.
head="$(git rev-parse HEAD)"
for sha in $full_shas; do
  [ "$sha" = "$head" ] || {
    echo "pre-push: pushing ${sha:0:10}, but the checkout is at ${head:0:10}; the checks would judge the wrong commit — check it out and push from there." >&2
    exit 1; }
done
if [ -n "$(git status --porcelain --untracked-files=no)" ]; then
  echo "pre-push: the work tree has uncommitted changes to tracked files; the checks would not judge what is pushed — refusing." >&2
  exit 1
fi

echo "pre-push: building the specification…"
BUILD_LOG="$(mktemp "${TMPDIR:-/tmp}/prepush-build.XXXXXX")" || BUILD_LOG=/dev/null
lake build >"$BUILD_LOG" 2>&1 || { echo "pre-push: BUILD FAILED — see $BUILD_LOG" >&2; exit 1; }

echo "pre-push: root checks…"
bash scripts/checks/run.sh || { echo "pre-push: ROOT CHECKS FAILED (above) — fix, or bypass deliberately." >&2; exit 1; }
echo "pre-push: build and root checks pass"
exit 0
